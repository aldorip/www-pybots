pybots.py 
from pybots import Bot, API, PDV

# Seu negócio automatizado
class NegocioInteligente:
    def __init__(self):
        self.bot     = Bot("telegram")
        self.api     = API("whatsapp")
        self.pdv     = PDV("supermercado")
        self.running = True

    async def run(self):
        # 24/7, sem parar
        while self.running:
            await self.bot.send_signal()
            await self.api.handle_msgs()
            await self.pdv.process_sale()

# Prontos para escalar com você
asyncio.run(NegocioInteligente().run())
